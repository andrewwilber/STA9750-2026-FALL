Introduction

While public transit plays an important role in reducing reliance on individual vehicle travel and its environmental consequences, its environmental performance varies considerably across states, metropolitan areas, and individual public transit agencies. This analysis provides an entry point for examining the environmental efficiency of public transit systems using ridership and energy consumption data from the Federal Transit Administration’s National Transit Database, coupled with electricity emissions data from the U.S. Energy Information Administration. This analysis is also used to select recipients of mock awards from the Green Transit Alliance’s 2025 Environmental Awards.

get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)

    state_formatted <- str_to_lower(state) |>
        str_replace_all("\\s", "")

    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)

    fs::dir_create(dir_name, recurse=TRUE)

    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |>
            req_url_path("electricity",
                         "state",
                         state_formatted) |>
            req_headers(
                `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>
            req_retry(
                max_tries = 4
            ) |>
            req_perform(file_name)

        resp_check_status(RESPONSE)
    }

    TABLE <- read_html(file_name) |>
        html_element("table") |>
        html_table() |>
        mutate(Item = str_to_lower(Item))

    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }

    CO2_MWh <- TABLE |>
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()

    PRIMARY <- TABLE |>
        filter(Item == "primary energy source") |>
        pull(Rank)

    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()

    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()

    data.frame(
        CO2_MWh               = CO2_MWh,
        primary_source        = PRIMARY,
        electricity_price_MWh = RATE * 10,
        generation_MWh        = GENERATION_MWh,
        state                 = state,
        abbreviation          = abbr
    )
}

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)

  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    library(httr2)

    request("https://www.transit.dot.gov/") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |>
      req_perform(NTD_ENERGY_FILE)
  }

  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
    cat("I was unable to download the NTD Energy File. Please try again.\n")
    stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
  }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |>
    select(-c(`Reporter Type`,
          `Reporting Module`,
          `Gallons of Other Fuel`,
          `Other Fuel Description`,
          `Bio-Diesel Description`)) |>
    mutate(across(-c(`Agency Name`,
                     `Mode`,
                     `TOS`),
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum),
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)

  # Create data directory if necessary
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse = TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){

    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query` = paste(
        "SELECT `_5_digit_ntd_id` AS `NTD_ID`",
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` AS `uza_code`",
        "`max_primary_uza_name` AS `uza_name`",
        "`max_primary_uza_area_sq_miles` AS `uza_sq_miles`",
        "`max_primary_uza_population` AS `uza_population`",
        "`max_service_area_sq_miles` AS `service_area_sq_miles`",
        "`max_service_area_population` AS `service_area_population`",
        "`questionable_record` AS `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` AS `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000",
        sep = ","
      )) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
      ) |>
      req_perform(NTD_SERVICE_FILE)
  }

  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
    cat("I was unable to download the NTD Service File. Please try again.\n")
    stop("Download failed. Delete data/mp02/2025_ntd_service.csv and try again.")
  }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(
      flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate)
    )
}

# Task 5: Join the three datasets

# Combine service records across types of service
# to create one record per agency and transportation mode.
service_by_mode <- NTD_SERVICE |>
  group_by(NTD_ID, mode) |>
  summarise(
    service_agency_name = first(AGENCY),
    state = first(state),
    number_of_states = n_distinct(state),
    total_upt = sum(total_upt, na.rm = TRUE),
    sum_passenger_miles = if (anyNA(sum_passenger_miles)) {
      NA_real_
    } else {
      sum(sum_passenger_miles)
    },
    .groups = "drop"
  )

# Standardize the agency identifier:
# Energy IDs are numeric; Service IDs are five-character strings.
energy_by_mode <- NTD_ENERGY |>
  mutate(
    NTD_ID = sprintf("%05d", as.integer(`NTD ID`))
  ) |>
  select(-`NTD ID`)

# Verify that agency/mode combinations are unique
stopifnot(
  !anyDuplicated(service_by_mode[c("NTD_ID", "mode")]),
  !anyDuplicated(energy_by_mode[c("NTD_ID", "Mode")]),
  all(service_by_mode$number_of_states == 1)
)

# Join fuel consumption with passenger service.
# Retain every agency/mode reporting energy consumption.
energy_service_joined <- energy_by_mode |>
  left_join(
    service_by_mode,
    by = c("NTD_ID", "Mode" = "mode"),
    relationship = "one-to-one"
  )

# Join the state-specific electricity emissions factor.
EMISSIONS_BASE <- energy_service_joined |>
  left_join(
    EIA_SEP_REPORT |>
      select(
        state_code = abbreviation,
        CO2_MWh
      ),

# Task 5: Calculate total CO2 emissions by agency and mode
# All intermediate emissions are measured in pounds of CO2.

EMISSIONS_TOTAL <- EMISSIONS_READY |>
  mutate(
    # CNG is reported in gasoline or diesel gallon equivalents.
    # Assume diesel equivalents for bus modes and gasoline
    # equivalents for other modes.
    cng_mmbtu_per_gallon_equivalent = if_else(
      Mode %in% c("MB", "CB", "RB"),
      0.138,
      0.114
    ),

    # EIA emissions factors: pounds CO2 per gallon
    diesel_co2_lb =
      `Gallons of Diesel Fuel` * 22.45,

    gasoline_co2_lb =
      `Gallons of Gasoline` * 18.73,

    bunker_co2_lb =
      `Gallons of Bunker Fuel` * 24.78,

    kerosene_co2_lb =
      `Gallons of Kerosene` * 21.78,

    lpg_co2_lb =
      `Gallons of Liquified Petroleum Gas` * 12.68,

    # EIA natural gas factor: 116.65 lb CO2 per MMBtu
    cng_co2_lb =
      `Gallons of Compressed Natural Gas` *
      cng_mmbtu_per_gallon_equivalent * 116.65,

    # EPA factors converted from kg CO2 to lb CO2
    biodiesel_co2_lb =
      `Gallons of Bio-Diesel` * 9.45 * 2.20462262,

    ethanol_co2_lb =
      `Gallons of Ethanol` * 5.75 * 2.20462262,

    lng_co2_lb =
      `Gallons of Liquified Nat Gas` * 4.50 * 2.20462262,

    # Hydrogen combustion produces no direct CO2.
    hydrogen_co2_lb =
      `Kilograms of Hydrogen` * 0,

    # Sum all combustion-fuel emissions.
    combustion_co2_lb =
      diesel_co2_lb +
      gasoline_co2_lb +
      bunker_co2_lb +
      kerosene_co2_lb +
      lpg_co2_lb +
      cng_co2_lb +
      biodiesel_co2_lb +
      ethanol_co2_lb +
      lng_co2_lb +
      hydrogen_co2_lb,

    # Add the electricity emissions calculated earlier.
    total_co2_lb =
      combustion_co2_lb + electricity_co2_lb,

      total_co2_lb / 2204.62262

```{r}
get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)

    state_formatted <- str_to_lower(state) |>
        str_replace_all("\\s", "")

    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)

    fs::dir_create(dir_name, recurse=TRUE)

    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |>
            req_url_path("electricity",
                         "state",
                         state_formatted) |>
            req_headers(
                `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>
            req_retry(
                max_tries = 4
            ) |>
            req_perform(file_name)

        resp_check_status(RESPONSE)
    }

    TABLE <- read_html(file_name) |>
        html_element("table") |>
        html_table() |>
        mutate(Item = str_to_lower(Item))

    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }

    CO2_MWh <- TABLE |>
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()

    PRIMARY <- TABLE |>
        filter(Item == "primary energy source") |>
        pull(Rank)

    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()

    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()

    data.frame(
        CO2_MWh               = CO2_MWh,
        primary_source        = PRIMARY,
        electricity_price_MWh = RATE * 10,
        generation_MWh        = GENERATION_MWh,
        state                 = state,
        abbreviation          = abbr
    )
}
```

```{r}
# Task 5: Calculate total CO2 emissions by agency and mode
# All intermediate emissions are measured in pounds of CO2.

EMISSIONS_TOTAL <- EMISSIONS_READY |>
  mutate(
    # CNG is reported in gasoline or diesel gallon equivalents.
    # Assume diesel equivalents for bus modes and gasoline
    # equivalents for other modes.
    cng_mmbtu_per_gallon_equivalent = if_else(
      Mode %in% c("MB", "CB", "RB"),
      0.138,
      0.114
    ),

    # EIA emissions factors: pounds CO2 per gallon
    diesel_co2_lb =
      `Gallons of Diesel Fuel` * 22.45,

    gasoline_co2_lb =
      `Gallons of Gasoline` * 18.73,

    bunker_co2_lb =
      `Gallons of Bunker Fuel` * 24.78,

    kerosene_co2_lb =
      `Gallons of Kerosene` * 21.78,

    lpg_co2_lb =
      `Gallons of Liquified Petroleum Gas` * 12.68,

    # EIA natural gas factor: 116.65 lb CO2 per MMBtu
    cng_co2_lb =
      `Gallons of Compressed Natural Gas` *
      cng_mmbtu_per_gallon_equivalent * 116.65,

    # EPA factors converted from kg CO2 to lb CO2
    biodiesel_co2_lb =
      `Gallons of Bio-Diesel` * 9.45 * 2.20462262,

    ethanol_co2_lb =
      `Gallons of Ethanol` * 5.75 * 2.20462262,

    lng_co2_lb =
      `Gallons of Liquified Nat Gas` * 4.50 * 2.20462262,

    # Hydrogen combustion produces no direct CO2.
    hydrogen_co2_lb =
      `Kilograms of Hydrogen` * 0,

    # Sum all combustion-fuel emissions.
    combustion_co2_lb =
      diesel_co2_lb +
      gasoline_co2_lb +
      bunker_co2_lb +
      kerosene_co2_lb +
      lpg_co2_lb +
      cng_co2_lb +
      biodiesel_co2_lb +
      ethanol_co2_lb +
      lng_co2_lb +
      hydrogen_co2_lb,

    # Add the electricity emissions calculated earlier.
    total_co2_lb =
      combustion_co2_lb + electricity_co2_lb,

    # Convert pounds to metric tons.
    total_co2_metric_tons =
      total_co2_lb / 2204.62262
  )

  all(EMISSIONS_TOTAL$total_co2_lb >= 0),
  isTRUE(all.equal(
    EMISSIONS_TOTAL$total_co2_lb,

)
```
